## 1. 按出现次数冒泡排序

给定数组：

```text
[4, 2, 4, 3, 2, 4, 5, 2]
```

使用冒泡排序，按照每个整数在数组中出现的次数从多到少排序；如果出现次数相同，则原整数数值较小的排在最终数组左侧。

例如：

```text
4 和 2 都出现了 3 次，3 和 5 各出现 1 次。
```

预期输出：

```text
2 2 2 4 4 4 3 5
```

请在给出的框架基础上补全排序函数或方法 `bubbleSortByFrequency`。

### C

```c
#include <stdio.h>

void bubbleSortByFrequency(int arr[], int length);

int main(void) {
    int arr[] = {4, 2, 4, 3, 2, 4, 5, 2};
    int length = sizeof(arr) / sizeof(arr[0]);

    bubbleSortByFrequency(arr, length);

    for (int i = 0; i < length; ++i) {
        if (i > 0) {
            printf(" ");
        }
        printf("%d", arr[i]);
    }
    printf("\n");
    return 0;
}
```

## 4. 含头节点的单链表增删改查

请实现带头节点的单链表。题目已经给出初始化、清空和打印代码，考生只需要实现插入、查找、修改和删除。

题目给出的 `main` 会打印操作后的链表结果。

### C

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

int init_list(Node **head) {
    *head = (Node *)malloc(sizeof(Node));
    if (*head == NULL) {
        return 0;
    }
    (*head)->value = 0;
    (*head)->next = NULL;
    return 1;
}

void destroy_list(Node **head) {
    if (head == NULL || *head == NULL) {
        return;
    }

    Node *current = *head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    *head = NULL;
}

void print_list(Node *head) {
    Node *current = head->next;
    while (current != NULL) {
        printf("%d", current->value);
        if (current->next != NULL) {
            printf(" ");
        }
        current = current->next;
    }
    printf("\n");
}

int insert_tail(Node *head, int value);
Node *find_value(Node *head, int value);
int update_value(Node *head, int old_value, int new_value);
int delete_value(Node *head, int value);

int main(void) {
    Node *head = NULL;
    if (!init_list(&head)) {
        return 1;
    }

    insert_tail(head, 10);
    insert_tail(head, 20);
    insert_tail(head, 30);

    if (find_value(head, 20) == NULL) {
        return 1;
    }

    update_value(head, 20, 25);
    delete_value(head, 10);
    print_list(head);

    destroy_list(&head);
    return 0;
}
```
